# Layer 4 降级引擎依赖（必需，用于 DuckDuckGo 搜索）
ddgs>=6.0.0

# 插件根 server.py（本地 stdio MCP server）的硬依赖，缺失时 server 直接退出
# 下限取本机实测版本 2.1.1；上限 <3 是因为 import 路径 mcp.server.mcpserver 是 2.x 专属，
# 3.x 若改名会把「装了包」的机器误诊成「没装包」（server.py 的 ImportError 提示无法区分这两种）
mcp>=2.1,<3

# ============================================================
# 全部引擎共用的传输层（必需）
# ============================================================

# curl_cffi — TLS/JA3 指纹伪装。原先写作"推荐安装、未装自动降级 urllib"，
# 2026-09-22 前后对照实测推翻这句：同一批 8 条 arXiv URL，urllib 全部 HTTP 406
# （连几十秒前刚返回过结果的查询也一样），装上后 7 条拿到真实字节。
# 缺它的后果不是"少个增强"，是 Lead 拿一套没有指纹的传输层去判"这个源不行"，
# 学术类与国内平台 claim 的升级路径从源头就断掉。--env-check 已把它列为硬缺失。
curl_cffi>=0.7.0
